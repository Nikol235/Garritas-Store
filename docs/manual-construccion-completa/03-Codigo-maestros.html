package pe.edu.upeu.garritas.catalogo.entity;

import jakarta.persistence.*;
import java.math.BigDecimal;
import java.util.HashSet;
import java.util.Set;
import lombok.*;

@Entity
@Table(name = "productos")
@Getter
@Setter
@Builder
@NoArgsConstructor
@AllArgsConstructor
public class Producto {

  @Version private Long version;

  @Id
  @GeneratedValue(strategy = GenerationType.IDENTITY)
  private Long id;

  @Column(name = "nombre", nullable = false, length = 100)
  private String nombre;

  @Column(name = "descripcion", length = 255)
  private String descripcion;

  @Column(name = "precio", nullable = false, precision = 10, scale = 2)
  private BigDecimal precio;

  @Column(name = "stock", nullable = false)
  private Integer stock;

  @Column(name = "activo", nullable = false)
  private Boolean activo;

  @ManyToOne(fetch = FetchType.LAZY)
  @JoinColumn(name = "id_categoria")
  private Categoria categoria;

  @Builder.Default
  @ManyToMany(fetch = FetchType.LAZY)
  @JoinTable(
      name = "producto_categoria",
      joinColumns = @JoinColumn(name = "id_producto"),
      inverseJoinColumns = @JoinColumn(name = "id_categoria"))
  private Set<Categoria> categorias = new HashSet<>();
}

package pe.edu.upeu.garritas.catalogo.mapper;

import java.util.Collections;
import java.util.Set;
import java.util.stream.Collectors;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;
import pe.edu.upeu.garritas.catalogo.dto.CategoriaResponse;
import pe.edu.upeu.garritas.catalogo.dto.ProductoRequest;
import pe.edu.upeu.garritas.catalogo.dto.ProductoResponse;
import pe.edu.upeu.garritas.catalogo.entity.Producto;

@Component
@RequiredArgsConstructor
public class ProductoMapper {

  private final CategoriaMapper categoriaMapper;

  public Producto toEntity(ProductoRequest request) {
    if (request == null) return null;
    return Producto.builder()
        .nombre(request.getNombre())
        .descripcion(request.getDescripcion())
        .precio(request.getPrecio())
        .stock(request.getStock())
        .activo(request.getActivo())
        .build();
  }

  public ProductoResponse toResponse(Producto producto) {
    if (producto == null) return null;
    Set<CategoriaResponse> categoriasResp =
        (producto.getCategorias() == null || producto.getCategorias().isEmpty())
            ? (producto.getCategoria() != null
                ? Set.of(categoriaMapper.toResponse(producto.getCategoria()))
                : Collections.emptySet())
            : producto.getCategorias().stream()
                .map(categoriaMapper::toResponse)
                .collect(Collectors.toSet());

    CategoriaResponse primaryCategoria =
        producto.getCategoria() != null
            ? categoriaMapper.toResponse(producto.getCategoria())
            : (!categoriasResp.isEmpty() ? categoriasResp.iterator().next() : null);

    return ProductoResponse.builder()
        .id(producto.getId())
        .nombre(producto.getNombre())
        .descripcion(producto.getDescripcion())
        .precio(producto.getPrecio())
        .stock(producto.getStock())
        .activo(producto.getActivo())
        .categoria(primaryCategoria)
        .categorias(categoriasResp)
        .build();
  }
}

import io.micrometer.core.instrument.MeterRegistry;

@Transactional
@Service
@RequiredArgsConstructor
public class ProductoService {

  private final ProductoRepository productoRepository;
  private final ProductoMapper productoMapper;
  private final CategoriaRepository categoriaRepository;
  private final MeterRegistry meterRegistry;

  public List<ProductoResponse> listar() {
    return productoRepository.findAllConCategoria().stream()
        .map(productoMapper::toResponse)
        .toList();
  }

  public Page<ProductoResponse> listarPaginado(String q, Long categoriaId, Pageable pageable) {
    return productoRepository.buscar(q, categoriaId, pageable).map(productoMapper::toResponse);
  }

  public ProductoResponse obtener(Long id) {
    return productoMapper.toResponse(buscarOFallar(id));
  }

  public ProductoResponse crear(ProductoRequest request) {
    Producto producto = productoMapper.toEntity(request);
    asignarCategoriasDesdeRequest(producto, request);
    ProductoResponse response = productoMapper.toResponse(productoRepository.save(producto));
    meterRegistry.counter("garritas_productos_creados_total").increment();
    return response;
  }

  public ProductoResponse actualizar(Long id, ProductoRequest request) {
    Producto producto = buscarOFallar(id);
    producto.setNombre(request.getNombre());
    producto.setDescripcion(request.getDescripcion());
    producto.setPrecio(request.getPrecio());
    producto.setStock(request.getStock());
    producto.setActivo(request.getActivo());
    asignarCategoriasDesdeRequest(producto, request);
    return productoMapper.toResponse(productoRepository.save(producto));
  }

  public void eliminar(Long id) {
    // Preserve inventory and historical references while asynchronous sagas finish.
    Producto producto = buscarOFallar(id);
    producto.setActivo(false);
    productoRepository.save(producto);
  }

  private void asignarCategoriasDesdeRequest(Producto producto, ProductoRequest request) {
    Set<Categoria> cats = new HashSet<>();
    if (request.getCategoriaIds() != null && !request.getCategoriaIds().isEmpty()) {
      cats.addAll(categoriaRepository.findAllById(request.getCategoriaIds()));
      if (cats.size() != request.getCategoriaIds().size())
        throw new org.springframework.web.server.ResponseStatusException(
            org.springframework.http.HttpStatus.BAD_REQUEST,
            "Una de las categorías seleccionadas no existe");
    } else if (request.getCategoriaIds() == null && request.getCategoriaId() != null) {
      cats.add(buscarCategoriaOFallar(request.getCategoriaId()));
    }
    producto.setCategorias(cats);
    producto.setCategoria(
        cats.stream().min(java.util.Comparator.comparing(Categoria::getId)).orElse(null));
  }

  private Producto buscarOFallar(Long id) {
    return productoRepository
        .findById(id)
        .orElseThrow(() -> new ResourceNotFoundException("Producto no encontrado: " + id));
  }

  private Categoria buscarCategoriaOFallar(Long categoriaId) {
    return categoriaRepository
        .findById(categoriaId)
        .orElseThrow(
            () -> new ResourceNotFoundException("Categoria no encontrada: " + categoriaId));
  }
}

package pe.edu.upeu.garritas.catalogo.messaging;

import java.math.BigDecimal;
import java.util.*;
import lombok.RequiredArgsConstructor;
import org.slf4j.*;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.kafka.annotation.KafkaListener;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;
import pe.edu.upeu.garritas.common.*;
import tools.jackson.databind.ObjectMapper;

@Component
@RequiredArgsConstructor
public class InventarioEvents {
  private static final Logger log = LoggerFactory.getLogger(InventarioEvents.class);
  private final JdbcTemplate jdbc;
  private final Inbox inbox;
  private final Outbox outbox;
  private final ObjectMapper json;

  @KafkaListener(topics = "${app.kafka.topic:garritas.ventas.v1}")
  @Transactional
  public void consume(String payload) {
    SagaEvent e = json.readValue(payload, SagaEvent.class);
    if (!Set.of("venta.creada", "pago.fallido").contains(e.type())) return;
    if (!inbox.claim(e)) return;
    try (var scope = MDC.putCloseable("traceId", e.traceId())) {
      if (e.type().equals("venta.creada")) reservar(e, payload);
      else liberar(e);
      log.info(
          "component=inventario ventaId={} eventType={} status=processed", e.ventaId(), e.type());
    }
  }

  private void reservar(SagaEvent e, String payload) {
    // Also deduplicate semantic replays carrying a different eventId.
    if (jdbc.update(
            "INSERT INTO reservas_stock(id_venta,estado,payload) VALUES (?,'NUEVA',?) ON CONFLICT"
                + " DO NOTHING",
            e.ventaId(),
            payload)
        == 0) return;
    var quantities = new TreeMap<Long, Integer>();
    e.items().forEach(i -> quantities.merge(i.productoId(), i.cantidad(), Math::addExact));
    boolean valid = true;
    // Lock in deterministic product order; validate every line before changing any stock.
    for (var item : quantities.entrySet()) {
      var rows =
          jdbc.queryForList(
              "SELECT stock,activo,precio FROM productos WHERE id=? FOR UPDATE", item.getKey());
      if (rows.isEmpty()) {
        valid = false;
        continue;
      }
      var p = rows.getFirst();
      if (!(Boolean) p.get("activo") || (Integer) p.get("stock") < item.getValue()) valid = false;
      for (var detail : e.items())
        if (detail.productoId().equals(item.getKey())
            && ((BigDecimal) p.get("precio")).compareTo(detail.precio()) != 0) valid = false;
    }
    if (valid)
      for (var item : quantities.entrySet())
        jdbc.update(
            "UPDATE productos SET stock=stock-?,version=version+1 WHERE id=?",
            item.getValue(),
            item.getKey());
    jdbc.update(
        "UPDATE reservas_stock SET estado=? WHERE id_venta=?",
        valid ? "RESERVADA" : "RECHAZADA",
        e.ventaId());
    outbox.append(e.next(valid ? "inventario.reservado" : "inventario.rechazado"));
  }

  private void liberar(SagaEvent e) {
    var rows =
        jdbc.queryForList(
            "SELECT estado,payload FROM reservas_stock WHERE id_venta=? FOR UPDATE", e.ventaId());
    if (rows.isEmpty()) throw new IllegalStateException("Reserva inexistente: " + e.ventaId());
    if (!rows.getFirst().get("estado").equals("RESERVADA")) return;
    var original = json.readValue(rows.getFirst().get("payload").toString(), SagaEvent.class);
    for (var item :
        original.items().stream().sorted(Comparator.comparing(SagaEvent.Item::productoId)).toList())
      jdbc.update(
          "UPDATE productos SET stock=stock+?,version=version+1 WHERE id=?",
          item.cantidad(),
          item.productoId());
    jdbc.update("UPDATE reservas_stock SET estado='LIBERADA' WHERE id_venta=?", e.ventaId());
    outbox.append(original.next("inventario.liberado"));
  }
}

import java.time.Instant;
import java.util.HashMap;
import java.util.Map;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

@RestControllerAdvice
@RequiredArgsConstructor
public class GlobalExceptionHandler {

  private final MeterRegistry meterRegistry;

  @ExceptionHandler(ResourceNotFoundException.class)
  public ResponseEntity<Map<String, Object>> handleNotFound(ResourceNotFoundException ex) {
    String recurso = ex.getMessage().split(" ")[0].toLowerCase();
    meterRegistry.counter("garritas_recursos_no_encontrados_total", "recurso", recurso).increment();

    Map<String, Object> body = new HashMap<>();
    body.put("timestamp", Instant.now().toString());
    body.put("status", HttpStatus.NOT_FOUND.value());
    body.put("error", "Not Found");
    body.put("message", ex.getMessage());
    return ResponseEntity.status(HttpStatus.NOT_FOUND).body(body);
  }

  @ExceptionHandler(MethodArgumentNotValidException.class)
  public ResponseEntity<Map<String, Object>> handleValidation(MethodArgumentNotValidException ex) {
    Map<String, Object> body = new HashMap<>();
    body.put("timestamp", Instant.now().toString());
    body.put("status", HttpStatus.BAD_REQUEST.value());
    body.put("error", "Bad Request");
    body.put(
        "message",
        ex.getBindingResult().getFieldErrors().stream()
            .map(e -> e.getField() + ": " + e.getDefaultMessage())
            .distinct()
            .collect(java.util.stream.Collectors.joining("; ")));
    return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
  }

  @ExceptionHandler(org.springframework.dao.DataIntegrityViolationException.class)
  public ResponseEntity<Map<String, Object>> handleConflict(
      org.springframework.dao.DataIntegrityViolationException ex) {
    return ResponseEntity.status(HttpStatus.CONFLICT)
        .body(
            Map.of(
                "message",
                "El registro ya existe o tiene información relacionada. Revisa los datos antes de"
                    + " continuar."));

package pe.edu.upeu.garritas.catalogo.filter;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.util.UUID;
import org.slf4j.MDC;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

@Component
public class CorrelationIdFilter extends OncePerRequestFilter {
  @org.springframework.beans.factory.annotation.Value("${spring.application.name}")
  private String application;

  @org.springframework.beans.factory.annotation.Value("${server.port}")
  private int port;

  public static final String TRACE_ID_HEADER = "X-Trace-ID";
  public static final String MDC_KEY = "traceId";

  @Override
  protected void doFilterInternal(
      HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
      throws ServletException, IOException {
    String traceId = request.getHeader(TRACE_ID_HEADER);
    if (traceId == null || !traceId.matches("[A-Za-z0-9._-]{1,100}")) {
      traceId = UUID.randomUUID().toString();
    }
    try {
      MDC.put(MDC_KEY, traceId);
      response.setHeader(TRACE_ID_HEADER, traceId);
      response.setHeader("X-Service-Instance", application + ":" + port);
      filterChain.doFilter(request, response);
    } finally {
      MDC.remove(MDC_KEY);
    }
  }
}

package pe.edu.upeu.garritas.institucion.service;

import java.util.*;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;
import pe.edu.upeu.garritas.common.CurrentUser;
import pe.edu.upeu.garritas.institucion.dto.*;
import pe.edu.upeu.garritas.institucion.entity.*;
import pe.edu.upeu.garritas.institucion.repository.*;

@Service
@RequiredArgsConstructor
@Transactional
public class PrestacionService {
  private final PrestacionRepository prestaciones;
  private final SedeRepository sedes;
  private final SedePrestacionRepository ofertas;
  private final org.springframework.jdbc.core.JdbcTemplate jdbc;

  public List<PrestacionResponse> listar() {
    return prestaciones.findAll().stream()
        .filter(p -> CurrentUser.admin() || disponible(p))
        .sorted(Comparator.comparing(Prestacion::getNombre))
        .map(this::responsePrestacion)
        .toList();
  }

  public PrestacionResponse guardar(Long id, PrestacionRequest r) {
    // Serialize graph edits across replicas so concurrent requests cannot introduce nested
    // packages.
    jdbc.queryForObject(
        "SELECT pg_advisory_xact_lock(hashtextextended(?,0))", Object.class, "garritas:paquetes");
    Prestacion p =
        id == null
            ? new Prestacion()
            : prestaciones.findById(id).orElseThrow(() -> error(404, "Prestación no encontrada"));
    var ids = r.componenteIds() == null ? java.util.Set.<Long>of() : r.componenteIds();
    if (!ids.isEmpty()) {
      if (ids.size() < 2) throw error(400, "Un paquete necesita al menos dos servicios distintos");
      if (id != null && ids.contains(id))
        throw error(400, "Un paquete no puede incluirse a sí mismo");
      var componentes = new java.util.HashSet<>(prestaciones.findAllById(ids));
      if (componentes.size() != ids.size())
        throw error(400, "Uno de los servicios del paquete no existe");
      if (componentes.stream().anyMatch(x -> !x.getActivo() || !x.getComponentes().isEmpty()))
        throw error(
            409,
            "Selecciona servicios individuales activos; no se permiten paquetes dentro de"
                + " paquetes");
      if (id != null
          && jdbc.queryForObject(
                  "SELECT count(*) FROM prestacion_componentes WHERE id_componente=?",
                  Long.class,
                  id)
              > 0) throw error(409, "Este servicio ya forma parte de otro paquete");
      p.setComponentes(componentes);
    } else {
      p.getComponentes().clear();
    }
    p.setNombre(r.nombre().strip());
    p.setDescripcion(r.descripcion());
    p.setActivo(r.activo());
    return responsePrestacion(prestaciones.saveAndFlush(p));
  }

  public void desactivar(Long id) {
    Prestacion p =
        prestaciones.findById(id).orElseThrow(() -> error(404, "Prestación no encontrada"));
    p.setActivo(false);
  }

  public List<OfertaResponse> ofertas(Long sede) {
    Sede s = sedes.findById(sede).orElseThrow(() -> error(404, "Sede no encontrada"));
    if (!s.getActivo() && !CurrentUser.admin()) throw error(404, "Sede no disponible");
    return ofertas.findBySedeId(sede).stream()
        .map(this::response)
        .filter(o -> CurrentUser.admin() || o.activo())
        .toList();
  }

  public OfertaResponse guardarOferta(Long sede, OfertaRequest r) {
    if (r.duracionMinutos() % 30 != 0)
      throw error(400, "La duración debe ser múltiplo de 30 minutos");
    Sede s = sedes.findById(sede).orElseThrow(() -> error(404, "Sede no encontrada"));
    Prestacion p =
        prestaciones
            .findById(r.prestacionId())
            .orElseThrow(() -> error(404, "Prestación no encontrada"));
    if (r.activo() && (!s.getActivo() || !disponible(p)))
      throw error(409, "Activa la sede y la prestación antes de ofrecerla");
    SedePrestacion o =
        ofertas.findBySedeIdAndPrestacionId(sede, p.getId()).orElseGet(SedePrestacion::new);
    o.setSede(s);
    o.setPrestacion(p);
    o.setPrecio(r.precio());
    o.setDuracionMinutos(r.duracionMinutos());
    o.setActivo(r.activo());
    return response(ofertas.saveAndFlush(o));
  }

  private OfertaResponse response(SedePrestacion o) {
    return new OfertaResponse(
        o.getId(),
        o.getSede().getId(),
        o.getPrestacion().getId(),
        o.getPrestacion().getNombre(),
        o.getPrecio(),
        o.getDuracionMinutos(),
        o.getActivo() && o.getSede().getActivo() && disponible(o.getPrestacion()),
        componentes(o.getPrestacion()));
  }

  private boolean disponible(Prestacion p) {
    return p.getActivo() && p.getComponentes().stream().allMatch(Prestacion::getActivo);
  }

  private java.util.List<ComponenteResponse> componentes(Prestacion p) {
    return p.getComponentes().stream()
        .sorted(java.util.Comparator.comparing(Prestacion::getId))
        .map(x -> new ComponenteResponse(x.getId(), x.getNombre()))
        .toList();
  }

  private ResponseStatusException error(int s, String m) {
    return new ResponseStatusException(HttpStatus.valueOf(s), m);
  }

  private PrestacionResponse responsePrestacion(Prestacion p) {
    return new PrestacionResponse(
        p.getId(), p.getNombre(), p.getDescripcion(), disponible(p), componentes(p));
  }
}

package pe.edu.upeu.garritas.institucion.exception;

import java.time.Instant;
import java.util.HashMap;
import java.util.Map;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

@RestControllerAdvice
public class GlobalExceptionHandler {

  @ExceptionHandler(ResourceNotFoundException.class)
  public ResponseEntity<Map<String, Object>> handleNotFound(ResourceNotFoundException ex) {
    Map<String, Object> body = new HashMap<>();
    body.put("timestamp", Instant.now().toString());
    body.put("status", HttpStatus.NOT_FOUND.value());
    body.put("error", "Not Found");
    body.put("message", ex.getMessage());
    return ResponseEntity.status(HttpStatus.NOT_FOUND).body(body);
  }

  @ExceptionHandler(MethodArgumentNotValidException.class)
  public ResponseEntity<Map<String, Object>> handleValidation(MethodArgumentNotValidException ex) {
    Map<String, Object> body = new HashMap<>();
    body.put("timestamp", Instant.now().toString());
    body.put("status", HttpStatus.BAD_REQUEST.value());
    body.put("error", "Bad Request");
    body.put(
        "message",
        ex.getBindingResult().getFieldErrors().stream()
            .map(e -> e.getField() + ": " + e.getDefaultMessage())
            .distinct()
            .collect(java.util.stream.Collectors.joining("; ")));
    return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
  }

  @ExceptionHandler(org.springframework.dao.DataIntegrityViolationException.class)
  public ResponseEntity<Map<String, Object>> handleConflict(
      org.springframework.dao.DataIntegrityViolationException ex) {
    return ResponseEntity.status(HttpStatus.CONFLICT)
        .body(
            Map.of(
                "message",
                "El registro ya existe o tiene información relacionada. Revisa los datos antes de"
                    + " continuar."));
  }
}

package pe.edu.upeu.garritas.institucion.filter;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.util.UUID;
import org.slf4j.MDC;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

@Component
public class CorrelationIdFilter extends OncePerRequestFilter {
  @org.springframework.beans.factory.annotation.Value("${spring.application.name}")
  private String application;

  @org.springframework.beans.factory.annotation.Value("${server.port}")
  private int port;

  public static final String TRACE_ID_HEADER = "X-Trace-ID";
  public static final String MDC_KEY = "traceId";

  @Override
  protected void doFilterInternal(
      HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
      throws ServletException, IOException {
    String traceId = request.getHeader(TRACE_ID_HEADER);
    if (traceId == null || !traceId.matches("[A-Za-z0-9._-]{1,100}")) {
      traceId = UUID.randomUUID().toString();
    }
    try {
      MDC.put(MDC_KEY, traceId);
      response.setHeader(TRACE_ID_HEADER, traceId);
      response.setHeader("X-Service-Instance", application + ":" + port);
      filterChain.doFilter(request, response);
    } finally {
      MDC.remove(MDC_KEY);
    }
  }
}

    if (r.sedeId() != null) {
      Reference sede =
          services.get("garritas-institucion-ms", "/api/v1/sedes/" + r.sedeId(), Reference.class);
      if (!Boolean.TRUE.equals(sede.activo()))
        throw new ResponseStatusException(HttpStatus.CONFLICT, "La sede no está activa");
    }
    Set<Long> ids = new HashSet<>(r.especialidadIds());
    List<Especialidad> found = especialidades.findAllById(ids);
    if (found.size() != ids.size() || found.stream().anyMatch(e -> !e.getActivo()))
      throw new ResponseStatusException(
          HttpStatus.BAD_REQUEST, "Selecciona especialidades existentes y activas");
    Profesional p = id == null ? new Profesional() : profesional(id);
    p.setNombre(r.nombre().trim());
    p.setEmail(r.email().trim().toLowerCase());
    p.setTelefono(r.telefono());
    p.setSedeId(r.sedeId());
    p.setActivo(r.activo());
    p.setEspecialidades(new HashSet<>(found));
    return response(profesionales.saveAndFlush(p));
  }

  public void eliminar(Long id) {
    Profesional p = profesional(id);
    p.setActivo(false);
    profesionales.save(p);
  }

  public List<Especialidad> especialidades() {
    return especialidades.findAll().stream()
        .filter(e -> CurrentUser.admin() || e.getActivo())
        .sorted(Comparator.comparing(Especialidad::getNombre))
        .toList();
  }

  public Especialidad guardarEspecialidad(Long id, EspecialidadRequest r) {
    Especialidad e =
        id == null
            ? new Especialidad()
            : especialidades
                .findById(id)
                .orElseThrow(
                    () ->
                        new ResponseStatusException(
                            HttpStatus.NOT_FOUND, "Especialidad no encontrada"));
    e.setNombre(r.nombre().trim());
    e.setDescripcion(r.descripcion());
    e.setActivo(r.activo());
    return especialidades.saveAndFlush(e);
  }

  public void eliminarEspecialidad(Long id) {
    Especialidad e =
        especialidades
            .findById(id)
            .orElseThrow(
                () ->
                    new ResponseStatusException(
                        HttpStatus.NOT_FOUND, "Especialidad no encontrada"));
    if (profesionales.countByEspecialidades_Id(id) > 0)
      throw new ResponseStatusException(
          HttpStatus.CONFLICT, "Primero quita la especialidad de sus profesionales");
    especialidades.delete(e);
  }

  private Profesional profesional(Long id) {
    return profesionales
        .findById(id)
        .orElseThrow(
            () -> new ResponseStatusException(HttpStatus.NOT_FOUND, "Profesional no encontrado"));
  }

  private ProfesionalResponse response(Profesional p) {
    return new ProfesionalResponse(
        p.getId(),
        p.getNombre(),
        p.getEmail(),
        p.getTelefono(),
        p.getSedeId(),
        p.getActivo(),
        p.getEspecialidades().stream()
            .sorted(Comparator.comparing(Especialidad::getNombre))
            .toList());
  }

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.util.UUID;
import org.slf4j.MDC;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

@Component
public class CorrelationIdFilter extends OncePerRequestFilter {
  @org.springframework.beans.factory.annotation.Value("${spring.application.name}")
  private String application;

  @org.springframework.beans.factory.annotation.Value("${server.port}")
  private int port;

  public static final String TRACE_ID_HEADER = "X-Trace-ID";
  public static final String MDC_KEY = "traceId";

  @Override
  protected void doFilterInternal(
      HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
      throws ServletException, IOException {
    String traceId = request.getHeader(TRACE_ID_HEADER);
    if (traceId == null || !traceId.matches("[A-Za-z0-9._-]{1,100}")) {
      traceId = UUID.randomUUID().toString();
    }
    try {
      MDC.put(MDC_KEY, traceId);
      response.setHeader(TRACE_ID_HEADER, traceId);
      response.setHeader("X-Service-Instance", application + ":" + port);
      filterChain.doFilter(request, response);
    } finally {
      MDC.remove(MDC_KEY);
    }
  }
}

package pe.edu.upeu.garritas.envio.service;

import java.util.*;
import lombok.RequiredArgsConstructor;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import org.springframework.web.server.ResponseStatusException;
import pe.edu.upeu.garritas.common.CurrentUser;
import pe.edu.upeu.garritas.envio.dto.*;
import pe.edu.upeu.garritas.envio.entity.*;
import pe.edu.upeu.garritas.envio.repository.*;

@Service
@RequiredArgsConstructor
@Transactional
public class LogisticaService {
  private final TransportistaRepository transportistas;
  private final ZonaRepository zonas;
  private final CoberturaRepository coberturas;

  public List<TransportistaResponse> transportistas() {
    return transportistas.findAll().stream()
        .filter(t -> CurrentUser.admin() || t.getActivo())
        .sorted(Comparator.comparing(Transportista::getNombre))
        .map(this::responseTransportista)
        .toList();
  }

  public List<ZonaResponse> zonas() {
    return zonas.findAll().stream()
        .filter(z -> CurrentUser.admin() || z.getActivo())
        .sorted(Comparator.comparing(Zona::getNombre))
        .map(this::responseZona)
        .toList();
  }

  public TransportistaResponse guardarTransportista(Long id, TransportistaRequest r) {
    Transportista t = id == null ? new Transportista() : transportista(id);
    t.setNombre(r.nombre().strip());
    t.setTelefono(r.telefono());
    t.setActivo(r.activo());
    return responseTransportista(transportistas.saveAndFlush(t));
  }

  public ZonaResponse guardarZona(Long id, ZonaRequest r) {
    Zona z = id == null ? new Zona() : zona(id);
    z.setNombre(r.nombre().strip());
    z.setCiudad(r.ciudad().strip());
    z.setActivo(r.activo());
    return responseZona(zonas.saveAndFlush(z));
  }

  public void borrarTransportista(Long id) {
    transportista(id).setActivo(false);
  }

  public void borrarZona(Long id) {
    zona(id).setActivo(false);
  }

  public List<CoberturaResponse> coberturas() {
    return coberturas.findAll().stream()
        .map(this::response)
        .filter(c -> CurrentUser.admin() || c.activo())
        .toList();
  }

  public CoberturaResponse cobertura(Long t, Long z) {
    CoberturaResponse c =
        response(
            coberturas
                .findByTransportistaIdAndZonaId(t, z)
                .orElseThrow(() -> error(404, "Sin cobertura para esa zona")));
    if (!c.activo()) throw error(409, "Cobertura no disponible");
    return c;
  }

  public CoberturaResponse guardarCobertura(CoberturaRequest r) {
    Transportista t = transportista(r.transportistaId());
    Zona z = zona(r.zonaId());
    if (r.activo() && (!t.getActivo() || !z.getActivo()))
      throw error(409, "Activa primero transportista y zona");
    Cobertura c =
        coberturas.findByTransportistaIdAndZonaId(t.getId(), z.getId()).orElseGet(Cobertura::new);
    c.setTransportista(t);
    c.setZona(z);
    c.setTarifa(r.tarifa());
    c.setDiasEstimados(r.diasEstimados());
    c.setActivo(r.activo());
    return response(coberturas.saveAndFlush(c));
  }

  public void borrarCobertura(Long id) {
    coberturas
        .findById(id)
        .orElseThrow(() -> error(404, "Cobertura no encontrada"))
        .setActivo(false);
  }

  private Transportista transportista(Long id) {
    return transportistas.findById(id).orElseThrow(() -> error(404, "Transportista no encontrado"));
  }

  private Zona zona(Long id) {
    return zonas.findById(id).orElseThrow(() -> error(404, "Zona no encontrada"));
  }

  private CoberturaResponse response(Cobertura c) {
    return new CoberturaResponse(
        c.getId(),
        c.getTransportista().getId(),
        c.getZona().getId(),
        c.getTransportista().getNombre(),
        c.getZona().getNombre(),
        c.getTarifa(),
        c.getDiasEstimados(),
        c.getActivo() && c.getTransportista().getActivo() && c.getZona().getActivo());
  }

  private ResponseStatusException error(int s, String m) {
    return new ResponseStatusException(HttpStatus.valueOf(s), m);
  }

  private TransportistaResponse responseTransportista(Transportista p) {
    return new TransportistaResponse(p.getId(), p.getNombre(), p.getTelefono(), p.getActivo());
  }

  private ZonaResponse responseZona(Zona p) {
    return new ZonaResponse(p.getId(), p.getNombre(), p.getCiudad(), p.getActivo());
  }
}

import java.time.Instant;
import java.util.HashMap;
import java.util.Map;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

@RestControllerAdvice
public class GlobalExceptionHandler {

  @ExceptionHandler(ResourceNotFoundException.class)
  public ResponseEntity<Map<String, Object>> handleNotFound(ResourceNotFoundException ex) {
    Map<String, Object> body = new HashMap<>();
    body.put("timestamp", Instant.now().toString());
    body.put("status", HttpStatus.NOT_FOUND.value());
    body.put("error", "Not Found");
    body.put("message", ex.getMessage());
    return ResponseEntity.status(HttpStatus.NOT_FOUND).body(body);
  }

  @ExceptionHandler(MethodArgumentNotValidException.class)
  public ResponseEntity<Map<String, Object>> handleValidation(MethodArgumentNotValidException ex) {
    Map<String, Object> body = new HashMap<>();
    body.put("timestamp", Instant.now().toString());
    body.put("status", HttpStatus.BAD_REQUEST.value());
    body.put("error", "Bad Request");
    body.put(
        "message",
        ex.getBindingResult().getFieldErrors().stream()
            .map(e -> e.getField() + ": " + e.getDefaultMessage())
            .distinct()
            .collect(java.util.stream.Collectors.joining("; ")));
    return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
  }

  @ExceptionHandler(org.springframework.dao.DataIntegrityViolationException.class)
  public ResponseEntity<Map<String, Object>> handleConflict(
      org.springframework.dao.DataIntegrityViolationException ex) {
    return ResponseEntity.status(HttpStatus.CONFLICT)
        .body(
            Map.of(
                "message",
                "El registro ya existe o tiene información relacionada. Revisa los datos antes de"
                    + " continuar."));
  }
}

package pe.edu.upeu.garritas.envio.filter;

import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.util.UUID;
import org.slf4j.MDC;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

@Component
public class CorrelationIdFilter extends OncePerRequestFilter {
  @org.springframework.beans.factory.annotation.Value("${spring.application.name}")
  private String application;

  @org.springframework.beans.factory.annotation.Value("${server.port}")
  private int port;

  public static final String TRACE_ID_HEADER = "X-Trace-ID";
  public static final String MDC_KEY = "traceId";

  @Override
  protected void doFilterInternal(
      HttpServletRequest request, HttpServletResponse response, FilterChain filterChain)
      throws ServletException, IOException {
    String traceId = request.getHeader(TRACE_ID_HEADER);
    if (traceId == null || !traceId.matches("[A-Za-z0-9._-]{1,100}")) {
      traceId = UUID.randomUUID().toString();
    }
    try {
      MDC.put(MDC_KEY, traceId);
      response.setHeader(TRACE_ID_HEADER, traceId);
      response.setHeader("X-Service-Instance", application + ":" + port);
      filterChain.doFilter(request, response);
    } finally {
      MDC.remove(MDC_KEY);
    }
  }
}

package pe.edu.upeu.garritas.envio;

import static org.junit.jupiter.api.Assertions.*;

import java.math.BigDecimal;
import java.util.*;
import org.junit.jupiter.api.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.security.authentication.UsernamePasswordAuthenticationToken;
import org.springframework.security.core.authority.SimpleGrantedAuthority;
import org.springframework.security.core.context.SecurityContextHolder;
import org.springframework.transaction.annotation.Transactional;
import pe.edu.upeu.garritas.envio.dto.*;
import pe.edu.upeu.garritas.envio.service.*;

@SpringBootTest
@Transactional
class LogisticaIntegrationTests {
  @Autowired LogisticaService service;

  void auth(String sub) {
    SecurityContextHolder.getContext()
        .setAuthentication(
            new UsernamePasswordAuthenticationToken(
                sub, "", List.of(new SimpleGrantedAuthority("ROLE_ADMIN"))));
  }

  @AfterEach
  void clear() {
    SecurityContextHolder.clearContext();
  }

  @Test
  void supportsManyCarriersManyZonesAndSoftDeactivation() {
    auth("prueba");
    String key = UUID.randomUUID().toString();
    var a = service.guardarTransportista(null, new TransportistaRequest("T1 " + key, "999", true));
    var b = service.guardarTransportista(null, new TransportistaRequest("T2 " + key, "999", true));
    var x = service.guardarZona(null, new ZonaRequest("Z1 " + key, "Lima", true));
    var y = service.guardarZona(null, new ZonaRequest("Z2 " + key, "Cusco", true));
    var original =
        service.guardarCobertura(new CoberturaRequest(a.id(), x.id(), BigDecimal.TEN, 1, true));
    service.guardarCobertura(new CoberturaRequest(b.id(), x.id(), BigDecimal.TEN, 2, true));
    service.guardarCobertura(new CoberturaRequest(a.id(), y.id(), BigDecimal.TEN, 3, true));
    var updated =
        service.guardarCobertura(
            new CoberturaRequest(a.id(), x.id(), new BigDecimal("12.00"), 2, true));
    assertEquals(original.id(), updated.id());
    assertEquals(new BigDecimal("12.00"), service.cobertura(a.id(), x.id()).tarifa());
    service.borrarZona(x.id());
    assertThrows(
        org.springframework.web.server.ResponseStatusException.class,
        () -> service.cobertura(a.id(), x.id()));
    assertTrue(service.cobertura(a.id(), y.id()).activo());
  }
}
